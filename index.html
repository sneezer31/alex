// （該当箇所のみ抜粋・修正済） /* ========== 読み取り・持ち主・マイグレーション洗浄 ========== */ function parseMsgs(text) { const arr = JSON.parse(text.replace(/^\uFEFF/, "")); if (!Array.isArray(arr)) throw new Error("配列ではありません"); const msgs = []; for (const x of arr) { if (x && typeof x === "object" && typeof x.text === "string") { let sw = x.search_word || null; // ★ 過去JSONの自動洗浄：thinking内の [検索] {"query":...} を抽出しプロパティへ格上げ if (!sw && x.thinking) { const match = x.thinking.match(/\[検索\]\s*(\{.*?\})/); if (match) { try { const sObj = JSON.parse(match[1]); sw = sObj.query || sObj.search_query || match[1]; } catch(e) { sw = match[1]; } } } // ★究極の防壁（ホワイトリスト方式） // attachedFilesや謎のメタデータはここで物理的に消滅します msgs.push({ time: String(x.time ?? ""), sender: String(x.sender ?? "?"), text: x.text, search_word: sw }); } } return { msgs, skipped: arr.length - msgs.length }; }
